3749: #[cfg(target_os = "linux")]
3750: fn tcp_connections_internal() -> Vec<TcpConnection> {
3751:     let mut connections = Vec::new();
3752:     if let Ok(content) = fs::read_to_string("/proc/net/tcp") {
3753:         for line in content.lines().skip(1) {
3754:             let fields: Vec<&str> = line.split_whitespace().collect();
3755:             if fields.len() < 4 {
3756:                 continue;
3757:             }
3758:             // /proc/net/tcp encodes each address as `<addr_hex>:<port_hex>`
3759:             // (8 hex chars + ':' + 4 hex chars for IPv4). split_once is the
3760:             // direct decoder. The earlier code split into a Vec, then called
3761:             // split_last (which returns a tuple of `(&last, rest)`), then
3762:             // tried to index `local_parts[0]` to access the original full
3763:             // string — that's a tuple-indexing type error
3764:             //   error[E0608]: cannot index into a value of type
3765:             //   `(&&str, &[&str])`
3766:             // and would have been semantically wrong even without the type
3767:             // error, since `local_parts.0` is the port, not the address+port
3768:             // string the slicing presumed.
3769:             if let Some((local_hex, local_port_hex)) = fields[1].split_once(':') {
3770:                 let local_addr = hex_to_ip(local_hex);
3771:                 if let Ok(port) = u16::from_str_radix(local_port_hex, 16) {
3772:                     let local = format!("{}:{}", local_addr, port);
3773: 
3774:                     if let Some((remote_hex, remote_port_hex)) = fields[2].split_once(':') {
3775:                         let remote_addr = hex_to_ip(remote_hex);
3776:                         if let Ok(port) = u16::from_str_radix(remote_port_hex, 16) {
3777:                             let remote = format!("{}:{}", remote_addr, port);
3778:                             let state = fields[3].to_string();
3779:                             connections.push(TcpConnection {
3780:                                 local_addr: local,
3781:                                 remote_addr: remote,
3782:                                 state,
3783:                                 pid: None,
3784:                             });
3785:                         }
3786:                     }
3787:                 }
3788:             }
3789:         }
3790:     }
3791:     connections
3792: }
