 271: fn validate_policy_safety(policy: &ContextualPolicySet) -> Result<(), RolloutError> {
 272:     let contains_allow_all = policy.rules.iter().any(|rule| {
 273:         rule.src == "*"
 274:             && rule.dst == "*"
 275:             && rule.protocol == Protocol::Any
 276:             && rule.action == RuleAction::Allow
 277:     });
 278:     if contains_allow_all {
 279:         return Err(RolloutError::UnsafeAllowAll);
 280:     }
 281:     Ok(())
 282: }
