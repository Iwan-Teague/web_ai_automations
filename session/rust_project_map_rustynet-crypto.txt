 548: fn store_in_windows_dpapi(key_id: &str, key_material: &[u8]) -> Result<(), CryptoError> {
 549:     let root = windows_dpapi_root()?;
 550:     validate_windows_dpapi_root(root.as_path())?;
 551:     let path = windows_dpapi_file_path(root.as_path(), key_id)?;
 552:     if path.exists() {
 553:         validate_windows_dpapi_file(path.as_path())?;
 554:     }
 555:     let mut protected = dpapi_protect(
 556:         key_material,
 557:         // LocalMachine scope is required so the daemon service (running as
 558:         // NT AUTHORITY\SYSTEM / LocalSystem) can decrypt key material that
 559:         // was stored by the bootstrap/install helper (running as the SSH
 560:         // user or an interactive admin). CurrentUser scope ties the blob
 561:         // to the encrypting user's master key, which is inaccessible to
 562:         // LocalSystem and causes CryptoError::DecryptionFailed at service
 563:         // startup (prepare_runtime_wireguard_key_material). NTFS ACLs on
 564:         // the key-custody directory (set by windows-runtime-acls-check and
 565:         // validated by validate_windows_dpapi_root/file) are the access
 566:         // boundary; DPAPI LocalMachine encryption provides at-rest
 567:         // protection against off-machine extraction.
 568:         WindowsDpapiScope::LocalMachine,
 569:         &format!("RustyNet key {key_id}"),
 570:     )
 571:     .map_err(|_| CryptoError::EncryptionFailed)?;
 572:     let write_result = write_windows_dpapi_blob(path.as_path(), &protected);
 573:     protected.zeroize();
 574:     write_result
 575: }

 641: fn write_windows_dpapi_blob(path: &Path, bytes: &[u8]) -> Result<(), CryptoError> {
 642:     use std::fs::OpenOptions;
 643:     use std::io::Write;
 644: 
 645:     let parent = path.parent().ok_or(CryptoError::Io)?;
 646:     validate_windows_dpapi_root(parent)?;
 647:     let candidate = parent.join(format!(
 648:         ".{}.{}.tmp",
 649:         path.file_name()
 650:             .and_then(|value| value.to_str())
 651:             .unwrap_or("key"),
 652:         std::process::id()
 653:     ));
 654:     if candidate.exists() {
 655:         let metadata =
 656:             std::fs::symlink_metadata(candidate.as_path()).map_err(|_| CryptoError::Io)?;
 657:         if metadata.file_type().is_symlink() || !metadata.is_file() {
 658:             return Err(CryptoError::PermissionDenied);
 659:         }
 660:         std::fs::remove_file(candidate.as_path()).map_err(|_| CryptoError::Io)?;
 661:     }
 662:     let mut file = OpenOptions::new()
 663:         .write(true)
 664:         .create_new(true)
 665:         .open(candidate.as_path())
 666:         .map_err(|_| CryptoError::Io)?;
 667:     file.write_all(bytes).map_err(|_| CryptoError::Io)?;
 668:     file.flush().map_err(|_| CryptoError::Io)?;
 669:     if path.exists() {
 670:         validate_windows_dpapi_file(path)?;
 671:         std::fs::remove_file(path).map_err(|_| CryptoError::Io)?;
 672:     }
 673:     std::fs::rename(candidate.as_path(), path).map_err(|_| CryptoError::Io)
 674: }

 799:     let payload_digest = hex_bytes(&Sha256::digest(payload));

 801:     Ok(ProviderAttestation {
 802:         provider_kind: provider.kind(),

 804:         payload_digest_hex: payload_digest,
 805:         signature_hex: hex_bytes(&signature),
 806:     })

 809: pub fn verify_provider_attestation(
 810:     provider: &dyn SigningProvider,
 811:     payload: &[u8],
 812:     attestation: &ProviderAttestation,
 813: ) -> Result<(), CryptoError> {
 814:     if attestation.provider_kind != provider.kind() {
 815:         return Err(CryptoError::AttestationVerificationFailed);
 816:     }
 817:     if attestation.key_identifier != provider.key_identifier() {
 818:         return Err(CryptoError::AttestationVerificationFailed);
 819:     }
 820: 
 821:     let expected_payload = hex_bytes(&Sha256::digest(payload));
 822:     if expected_payload != attestation.payload_digest_hex {
 823:         return Err(CryptoError::AttestationVerificationFailed);
 824:     }
 825: 
 826:     let signature = hex_decode(attestation.signature_hex.as_str())?;
 827:     provider.verify_attestation(payload, &signature)
 828: }

 830: fn hex_bytes(bytes: &[u8]) -> String {
 831:     let mut out = String::with_capacity(bytes.len() * 2);
 832:     for byte in bytes {
 833:         out.push_str(&format!("{byte:02x}"));
 834:     }
 835:     out
 836: }

 838: fn hex_decode(value: &str) -> Result<Vec<u8>, CryptoError> {
 839:     let bytes = value.as_bytes();
 840:     if bytes.is_empty() || (bytes.len() & 1) != 0 {
 841:         return Err(CryptoError::AttestationVerificationFailed);
 842:     }
 843: 
 844:     let mut out = Vec::with_capacity(bytes.len() / 2);
 845:     for pair in bytes.chunks_exact(2) {
 846:         let hi = (pair[0] as char)
 847:             .to_digit(16)
 848:             .ok_or(CryptoError::AttestationVerificationFailed)?;
 849:         let lo = (pair[1] as char)
 850:             .to_digit(16)
 851:             .ok_or(CryptoError::AttestationVerificationFailed)?;
 852:         out.push(((hi << 4) | lo) as u8);
 853:     }
 854:     Ok(out)
 855: }

 858:     let mut rng = rand::rng();
 859:     let mut salt = [0u8; 16];
 860:     let mut nonce = [0u8; 24];

 862:     rng.fill_bytes(&mut nonce);

 866: pub fn encrypt_private_key_envelope(
 867:     plaintext: &[u8],
 868:     passphrase: &str,
 869:     salt: [u8; 16],
 870:     nonce: [u8; 24],
 871: ) -> Result<EncryptedKeyBlob, CryptoError> {
 872:     let mut key = [0u8; 32];
 873:     Argon2::default()
 874:         .hash_password_into(passphrase.as_bytes(), &salt, &mut key)
 875:         .map_err(|_| CryptoError::KdfFailed)?;
 876: 
 877:     let cipher = XChaCha20Poly1305::new((&key).into());
 878:     let ciphertext = match cipher.encrypt(XNonce::from_slice(&nonce), plaintext) {
 879:         Ok(value) => value,
 880:         Err(_) => {
 881:             key.fill(0);
 882:             return Err(CryptoError::EncryptionFailed);
 883:         }
 884:     };
 885: 
 886:     key.fill(0);
 887: 
 888:     Ok(EncryptedKeyBlob {
 889:         salt,
 890:         nonce,
 891:         ciphertext,
 892:     })
 893: }
 894: 
 895: pub fn decrypt_private_key_envelope(
 896:     blob: &EncryptedKeyBlob,
 897:     passphrase: &str,
 898: ) -> Result<Vec<u8>, CryptoError> {
 899:     let mut key = [0u8; 32];
 900:     Argon2::default()
 901:         .hash_password_into(passphrase.as_bytes(), &blob.salt, &mut key)
 902:         .map_err(|_| CryptoError::KdfFailed)?;
 903: 
 904:     let cipher = XChaCha20Poly1305::new((&key).into());
 905:     let plaintext = match cipher.decrypt(XNonce::from_slice(&blob.nonce), blob.ciphertext.as_ref())
 906:     {
 907:         Ok(value) => value,
 908:         Err(_) => {
 909:             key.fill(0);
 910:             return Err(CryptoError::DecryptionFailed);
 911:         }
 912:     };
 913: 
 914:     key.fill(0);
 915: 
 916:     Ok(plaintext)
 917: }

 919: pub fn write_encrypted_key_file(
 920:     directory: &Path,
 921:     file: &Path,
 922:     plaintext: &[u8],
 923:     passphrase: &str,
 924:     policy: KeyCustodyPermissionPolicy,
 925: ) -> Result<(), CryptoError> {
 926:     let (salt, nonce) = generate_key_custody_material();
 927:     let blob = encrypt_private_key_envelope(plaintext, passphrase, salt, nonce)?;
 928:     let encoded = encode_encrypted_blob(&blob);
 929: 
 930:     if directory.exists() {
 931:         let directory_link_metadata =
 932:             std::fs::symlink_metadata(directory).map_err(|_| CryptoError::Io)?;
 933:         if directory_link_metadata.file_type().is_symlink() || !directory_link_metadata.is_dir() {
 934:             return Err(CryptoError::PermissionDenied);
 935:         }
 936:     }
 937: 
 938:     std::fs::create_dir_all(directory).map_err(|_| CryptoError::Io)?;
 939:     if file.parent() != Some(directory) {
 940:         return Err(CryptoError::Io);

 947:             directory,

 949:         )
 950:         .map_err(|_| CryptoError::Io)?;
 951:         std::fs::set_permissions(
 952:             file,
 953:             std::fs::Permissions::from_mode(policy.required_file_mode),
 954:         )
 955:         .map_err(|_| CryptoError::Io)?;
 956:     }
 957:     validate_key_custody_permissions(directory, file, policy)?;
 958:     Ok(())

 961: fn write_atomic_encrypted_key_file(
 962:     path: &Path,
 963:     bytes: &[u8],
 964:     _mode: u32,
 965: ) -> Result<(), CryptoError> {
 966:     use std::fs::OpenOptions;
 967:     use std::io::Write;
 968: 
 969:     let parent = path.parent().ok_or(CryptoError::Io)?;
 970:     let temp = temp_path_for(path);
 971: 
 972:     let mut options = OpenOptions::new();
 973:     options.write(true).create_new(true);
 974:     #[cfg(unix)]
 975:     {
 976:         use std::os::unix::fs::OpenOptionsExt;
 977:         options.mode(_mode);
 978:     }
 979: 
 980:     let mut file = options.open(&temp).map_err(|_| CryptoError::Io)?;
 981:     if file.write_all(bytes).is_err() {
 982:         let _ = std::fs::remove_file(&temp);
 983:         return Err(CryptoError::Io);
 984:     }
 985:     if file.sync_all().is_err() {
 986:         let _ = std::fs::remove_file(&temp);
 987:         return Err(CryptoError::Io);
 988:     }
 989:     if std::fs::rename(&temp, path).is_err() {
 990:         let _ = std::fs::remove_file(&temp);
 991:         return Err(CryptoError::Io);
 992:     }
 993: 
 994:     // Directory fsync is a no-op on Windows: FlushFileBuffers on a directory
 995:     // handle requires special access flags not provided by File::open, and the
 996:     // durability guarantee is enforced by the rename above.
 997:     #[cfg(unix)]
 998:     {
 999:         let parent_dir = std::fs::File::open(parent).map_err(|_| CryptoError::Io)?;
1000:         parent_dir.sync_all().map_err(|_| CryptoError::Io)?;
1001:     }
1002:     Ok(())
1003: }

1009:         .map(|duration| duration.as_nanos())
1010:         .unwrap_or(0);

1027: fn encode_encrypted_blob(blob: &EncryptedKeyBlob) -> Vec<u8> {
1028:     let mut out = Vec::with_capacity(16 + 24 + 4 + blob.ciphertext.len());
1029:     out.extend_from_slice(&blob.salt);
1030:     out.extend_from_slice(&blob.nonce);
1031:     out.extend_from_slice(&(blob.ciphertext.len() as u32).to_be_bytes());
1032:     out.extend_from_slice(&blob.ciphertext);
1033:     out
1034: }

1036: fn decode_encrypted_blob(bytes: &[u8]) -> Result<EncryptedKeyBlob, CryptoError> {
1037:     if bytes.len() < 44 {
1038:         return Err(CryptoError::InvalidLength);
1039:     }
1040: 
1041:     let mut salt = [0u8; 16];
1042:     salt.copy_from_slice(&bytes[0..16]);
1043:     let mut nonce = [0u8; 24];
1044:     nonce.copy_from_slice(&bytes[16..40]);
1045: 
1046:     let mut length_bytes = [0u8; 4];
1047:     length_bytes.copy_from_slice(&bytes[40..44]);
1048:     let ciphertext_len = u32::from_be_bytes(length_bytes) as usize;
1049:     if bytes.len() != 44 + ciphertext_len {
1050:         return Err(CryptoError::InvalidLength);
1051:     }
1052: 
1053:     Ok(EncryptedKeyBlob {
1054:         salt,
1055:         nonce,
1056:         ciphertext: bytes[44..].to_vec(),
1057:     })
1058: }

1077:     file: &Path,
1078:     policy: KeyCustodyPermissionPolicy,
1079: ) -> Result<(), CryptoError> {
1080:     #[cfg(unix)]
1081:     {
1082:         use std::os::unix::fs::PermissionsExt;
1083: 
1084:         let directory_link_metadata =
1085:             std::fs::symlink_metadata(directory).map_err(|_| CryptoError::Io)?;
1086:         if directory_link_metadata.file_type().is_symlink() || !directory_link_metadata.is_dir() {
1087:             return Err(CryptoError::PermissionDenied);
1088:         }
1089:         let file_link_metadata = std::fs::symlink_metadata(file).map_err(|_| CryptoError::Io)?;
1090:         if file_link_metadata.file_type().is_symlink() || !file_link_metadata.is_file() {
1091:             return Err(CryptoError::PermissionDenied);
1092:         }
1093: 
1094:         let directory_metadata = std::fs::metadata(directory).map_err(|_| CryptoError::Io)?;
1095:         let file_metadata = std::fs::metadata(file).map_err(|_| CryptoError::Io)?;
1096: 
1097:         let directory_mode = directory_metadata.permissions().mode() & 0o777;
1098:         let file_mode = file_metadata.permissions().mode() & 0o777;
