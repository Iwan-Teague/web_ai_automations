   3: //! Rustynet Relay Daemon
   4: //!
   5: //! A production relay server that enables NAT traversal for Rustynet nodes
   6: //! when direct UDP connectivity is unavailable.
   7: //!
   8: //! # Architecture
   9: //!
  10: //! The relay uses **allocated-port demultiplexing**:
  11: //! - A control port receives RelayHello messages and allocates sessions
  12: //! - Each session gets a unique allocated port for ciphertext forwarding
  13: //! - Inbound packets on allocated ports are forwarded to the paired session
  14: //!
  15: //! # Security Model
  16: //!
  17: //! - Ciphertext-only: relay never sees plaintext
  18: //! - Signed tokens: ed25519 signatures from control plane
  19: //! - Constant-time auth: no timing side channels
  20: //! - Replay protection: nonce tracking
  21: //! - Session binding: tokens bound to (node_id, peer_node_id, relay_id)
  22: //! - Rate limiting: per-node packet and hello rate limits
  23: //!
  24: //! # Usage
  25: //!
  26: //! ```bash
  27: //! rustynet-relay --bind 0.0.0.0:4500 \
  28: //!                --relay-id "relay-us-east-1" \
  29: //!                --verifier-key /path/to/control-verifier.pub \
  30: //!                --port-range 50000-59999
  31: //! ```

 150:         /// Allocates a new port for a session.
 151:         async fn allocate_port(&self) -> Result<(u16, UdpSocket), String> {
 152:             let mut next = self.next_port.write().await;
 153:             let sockets = self.allocated_sockets.read().await;
 154: 
 155:             let range_size =
 156:                 (self.config.port_range_end - self.config.port_range_start + 1) as usize;
 157:             let mut attempts = 0;
 158: 
 159:             loop {
 160:                 if attempts >= range_size {
 161:                     return Err("no available ports in range".to_string());
 162:                 }
 163: 
 164:                 let port = *next;
 165:                 *next = if *next >= self.config.port_range_end {
 166:                     self.config.port_range_start
 167:                 } else {
 168:                     *next + 1
 169:                 };
 170: 
 171:                 if sockets.contains_key(&port) {
 172:                     attempts += 1;
 173:                     continue;
 174:                 }

 177:                 let addr = SocketAddr::new(self.config.bind_addr.ip(), port);

 179:                     Ok(socket) => return Ok((port, socket)),
 180:                     Err(_) => {
 181:                         attempts += 1;
 182:                         continue;
 183:                     }
 184:                 }

 430:     fn parse_relay_hello(data: &[u8]) -> Result<rustynet_relay::transport::RelayHello, String> {
 431:         if data.is_empty() || data[0] != RELAY_HELLO_MSG_TYPE {
 432:             return Err("not a hello message".to_string());
 433:         }
 434: 
 435:         let mut pos = 1;
 436: 
 437:         // Node ID
 438:         if pos + 2 > data.len() {
 439:             return Err("truncated node_id length".to_string());
 440:         }
 441:         let node_id_len = u16::from_be_bytes([data[pos], data[pos + 1]]) as usize;
 442:         pos += 2;
 443: 
 444:         if pos + node_id_len > data.len() {
 445:             return Err("truncated node_id".to_string());
 446:         }
 447:         let node_id =

 449:         pos += node_id_len;

 451:         // Peer node ID
 452:         if pos + 2 > data.len() {
 453:             return Err("truncated peer_node_id length".to_string());
 454:         }

 459:             return Err("truncated peer_node_id".to_string());

 462:             .map_err(|e| e.to_string())?;

 464: 
 465:         // Token
 466:         if pos + 2 > data.len() {
 467:             return Err("truncated token length".to_string());

 485:     /// Parses a RelaySessionToken from wire format.
 486:     fn parse_relay_token(data: &[u8]) -> Result<rustynet_control::RelaySessionToken, String> {
 487:         use rustynet_control::RelaySessionToken;
 488: 
 489:         let mut pos = 0;
 490: 
 491:         // Version
 492:         if pos >= data.len() {
 493:             return Err("missing version".to_string());
 494:         }
 495:         let version = data[pos];
 496:         if version != 1 {
 497:             return Err(format!("unsupported token version: {version}"));
 498:         }
 499:         pos += 1;
 500: 
 501:         // Node ID
 502:         if pos + 2 > data.len() {
 503:             return Err("truncated token node_id length".to_string());
 504:         }
 505:         let node_id_len = u16::from_be_bytes([data[pos], data[pos + 1]]) as usize;
 506:         pos += 2;
 507: 
 508:         if pos + node_id_len > data.len() {
 509:             return Err("truncated token node_id".to_string());
 510:         }
 511:         let node_id = String::from_utf8(data[pos..pos + node_id_len].to_vec())
 512:             .map_err(|e| format!("invalid token node_id: {e}"))?;
 513:         pos += node_id_len;
 514: 
 515:         // Peer node ID
 516:         if pos + 2 > data.len() {
 517:             return Err("truncated token peer_node_id length".to_string());
 518:         }
 519:         let peer_node_id_len = u16::from_be_bytes([data[pos], data[pos + 1]]) as usize;
 520:         pos += 2;
 521: 
 522:         if pos + peer_node_id_len > data.len() {
 523:             return Err("truncated token peer_node_id".to_string());
 524:         }
 525:         let peer_node_id = String::from_utf8(data[pos..pos + peer_node_id_len].to_vec())
 526:             .map_err(|e| format!("invalid token peer_node_id: {e}"))?;
 527:         pos += peer_node_id_len;
 528: 
 529:         // Relay ID (16 bytes)
 530:         if pos + 16 > data.len() {
 531:             return Err("truncated relay_id".to_string());
 532:         }
 533:         let relay_id: [u8; 16] = data[pos..pos + 16]
 534:             .try_into()
 535:             .map_err(|_| "invalid relay_id")?;
 536:         pos += 16;
 537: 
 538:         // Scope
 539:         if pos + 2 > data.len() {
 540:             return Err("truncated scope length".to_string());
 541:         }
 542:         let scope_len = u16::from_be_bytes([data[pos], data[pos + 1]]) as usize;
 543:         pos += 2;
 544: 
 545:         if pos + scope_len > data.len() {
 546:             return Err("truncated scope".to_string());
 547:         }
 548:         let scope = String::from_utf8(data[pos..pos + scope_len].to_vec())
 549:             .map_err(|e| format!("invalid scope: {e}"))?;
 550:         pos += scope_len;
 551: 
 552:         // Timestamps
 553:         if pos + 16 > data.len() {
 554:             return Err("truncated timestamps".to_string());
 555:         }
 556:         let issued_at_unix = u64::from_be_bytes(data[pos..pos + 8].try_into().unwrap());
 557:         pos += 8;
 558:         let expires_at_unix = u64::from_be_bytes(data[pos..pos + 8].try_into().unwrap());
 559:         pos += 8;
 560: 
 561:         // Nonce (16 bytes)
 562:         if pos + 16 > data.len() {
 563:             return Err("truncated nonce".to_string());
 564:         }
 565:         let nonce: [u8; 16] = data[pos..pos + 16]
 566:             .try_into()
 567:             .map_err(|_| "invalid nonce")?;
 568:         pos += 16;
 569: 
 570:         // Signature (64 bytes)
 571:         if pos + 64 > data.len() {
 572:             return Err("truncated signature".to_string());
 573:         }
 574:         let signature: [u8; 64] = data[pos..pos + 64]
 575:             .try_into()
 576:             .map_err(|_| "invalid signature")?;
 577: 
 578:         Ok(RelaySessionToken {
 579:             node_id,
 580:             peer_node_id,
 581:             relay_id,
 582:             scope,
 583:             issued_at_unix,
 584:             expires_at_unix,
 585:             nonce,
 586:             signature,
 587:         })
 588:     }

 607:     /// Parses command-line arguments into RelayConfig.
 608:     pub fn parse_args() -> Result<RelayConfig, String> {
 609:         let args: Vec<String> = std::env::args().collect();
 610:         let mut config = RelayConfig::default();
 611: 
 612:         let mut i = 1;
 613:         while i < args.len() {
 614:             match args[i].as_str() {
 615:                 "--bind" => {
 616:                     i += 1;
 617:                     if i >= args.len() {
 618:                         return Err("--bind requires an argument".to_string());
 619:                     }
 620:                     config.bind_addr = args[i]
 621:                         .parse()
 622:                         .map_err(|e| format!("invalid bind address: {e}"))?;
 623:                 }
 624:                 "--relay-id" => {
 625:                     i += 1;
 626:                     if i >= args.len() {
 627:                         return Err("--relay-id requires an argument".to_string());
 628:                     }
 629:                     let id_str = &args[i];
 630:                     // Hash the string to get a 16-byte relay ID
 631:                     use sha2::{Digest, Sha256};
 632:                     let hash = Sha256::digest(id_str.as_bytes());
 633:                     config.relay_id.copy_from_slice(&hash[..16]);
 634:                 }
 635:                 "--verifier-key" => {
 636:                     i += 1;
 637:                     if i >= args.len() {
 638:                         return Err("--verifier-key requires an argument".to_string());
 639:                     }
 640:                     config.verifier_key_path = args[i].clone();
 641:                 }
 642:                 "--port-range" => {
 643:                     i += 1;
 644:                     if i >= args.len() {
 645:                         return Err("--port-range requires an argument".to_string());
 646:                     }
 647:                     let parts: Vec<&str> = args[i].split('-').collect();
 648:                     if parts.len() != 2 {
 649:                         return Err("--port-range must be START-END".to_string());
 650:                     }
 651:                     config.port_range_start = parts[0]
 652:                         .parse()
 653:                         .map_err(|e| format!("invalid port range start: {e}"))?;
 654:                     config.port_range_end = parts[1]
 655:                         .parse()
 656:                         .map_err(|e| format!("invalid port range end: {e}"))?;
 657:                 }
 658:                 "--max-sessions-per-node" => {
 659:                     i += 1;
 660:                     if i >= args.len() {
 661:                         return Err("--max-sessions-per-node requires an argument".to_string());
 662:                     }
 663:                     config.max_sessions_per_node = args[i]
 664:                         .parse()
 665:                         .map_err(|e| format!("invalid max sessions: {e}"))?;
 666:                 }
 667:                 "--help" | "-h" => {
 668:                     eprintln!(
 669:                         "Usage: rustynet-relay [OPTIONS]\n\n\
 670:                         Options:\n  \
 671:                           --bind <ADDR>              UDP bind address (default: 0.0.0.0:4500)\n  \
 672:                           --relay-id <ID>            Relay identifier string\n  \
 673:                           --verifier-key <PATH>      Path to control verifier public key\n  \
 674:                           --port-range <START-END>   Port range for allocations (default: 50000-59999)\n  \
 675:                           --max-sessions-per-node <N> Max sessions per node (default: 8)\n  \
 676:                           --help                     Show this help"
 677:                     );
 678:                     std::process::exit(0);
 679:                 }
 680:                 arg => {
 681:                     return Err(format!("unknown argument: {arg}"));
 682:                 }
 683:             }
 684:             i += 1;
 685:         }
 686: 
 687:         if config.verifier_key_path.is_empty() {
 688:             return Err("--verifier-key is required".to_string());
 689:         }
 690: 
 691:         if config.relay_id == [0; 16] {
 692:             return Err("--relay-id is required".to_string());
 693:         }
 694: 
 695:         Ok(config)
 696:     }

  48:     fn new(max_pps: u64, max_bps: u64) -> Self {
  49:         Self {
  50:             max_pps,
  51:             max_bps,
  52:             packet_tokens: max_pps as f64,

  58:     fn check_and_consume(&mut self, packets: u64, bits: usize) -> bool {
  59:         self.refill();
  60: 
  61:         if self.packet_tokens >= packets as f64 && self.bit_tokens >= bits as f64 {
  62:             self.packet_tokens -= packets as f64;
  63:             self.bit_tokens -= bits as f64;
  64:             true
  65:         } else {
  66:             false
  67:         }
  68:     }

  72:         let elapsed = now.duration_since(self.last_refill).as_secs_f64();
  73:         self.last_refill = now;
  74: 
  75:         self.packet_tokens =
  76:             (self.packet_tokens + elapsed * self.max_pps as f64).min(self.max_pps as f64);
  77: 
  78:         self.bit_tokens =
  79:             (self.bit_tokens + elapsed * self.max_bps as f64).min(self.max_bps as f64);
  80:     }

   9: //!
  10: //! - **Ciphertext-only relay**: payloads are forwarded byte-for-byte without
  11: //!   inspection.  The relay never sees plaintext.
  12: //! - **Signed tokens**: every session requires a control-plane-issued
  13: //!   [`RelaySessionToken`] with a valid ed25519 signature.
  14: //! - **Constant-time auth**: all secret-field comparisons (`node_id`,
  15: //!   `peer_node_id`, `relay_id`) use `subtle::ConstantTimeEq`.
  16: //! - **Replay protection**: each token nonce is remembered for the duration of
  17: //!   the max TTL window; replayed nonces are rejected.
  18: //! - **Relay-binding**: tokens carry the intended relay's `relay_id`; the
  19: //!   relay verifies this matches its own identity.
  20: //! - **Scope enforcement**: tokens must carry `scope=forward_ciphertext_only`.
  21: //! - **Bounded resources**: per-node session caps, per-node packet rate limits,
  22: //!   per-node hello rate limits, maximum packet size, idle/half-open timeouts.
  23:

  35: /// Maximum TTL we accept from a token's `issued_at` → `expires_at` span.
  36: const MAX_RELAY_TTL_SECS: u64 = 120;
  37: /// How long we keep a half-open (un-paired) session before evicting it.
  38: const HALF_OPEN_SESSION_TIMEOUT_SECS: u64 = 60;
  39: /// How long an idle (no packets) session may remain open.
  40: const IDLE_SESSION_TIMEOUT_SECS: u64 = 30;
  41: /// Maximum forwarded-payload size.  Packets larger than this are dropped
  42: /// silently rather than returned as an error (to avoid amplification).
  43: const MAX_PACKET_SIZE_BYTES: usize = 65_536;
  44: /// How long we retain nonces in the replay-prevention store (must cover the
  45: /// full TTL window so a nonce cannot be recycled within its validity period).
  46: const NONCE_RETENTION_SECS: u64 = MAX_RELAY_TTL_SECS * 2;
  47: /// Maximum `handle_hello` calls accepted per node within a one-second window
  48: /// before the hello itself is rejected (separate from packet rate limiting).
  49: const MAX_HELLOS_PER_NODE_PER_SEC: u32 = 5;

 107: pub struct RelayTransport {
 108:     /// This relay's own identifier.  Every accepted token must have a matching
 109:     /// `relay_id`; tokens issued for a different relay are rejected.
 110:     relay_id: [u8; 16],
 111:     sessions: HashMap<SessionId, RelaySession>,
 112:     node_pair_index: HashMap<(String, String), SessionId>,
 113:     rate_limiter: RateLimiter,
 114:     control_verifier_key: VerifyingKey,
 115:     nonce_store: NonceStore,
 116:     hello_limiter: HelloLimiter,
 117:     clock_skew_tolerance_secs: u64,
 118:     max_sessions_per_node: usize,
 119: }

 121: impl RelayTransport {
 122:     pub fn new(
 123:         relay_id: [u8; 16],
 124:         control_verifier_key: VerifyingKey,
 125:         max_sessions_per_node: usize,
 126:         clock_skew_tolerance_secs: u64,
 127:     ) -> Self {
 128:         Self {
 129:             relay_id,
 130:             sessions: HashMap::new(),
 131:             node_pair_index: HashMap::new(),
 132:             rate_limiter: RateLimiter::default(),
 133:             control_verifier_key,
 134:             nonce_store: NonceStore::default(),
 135:             hello_limiter: HelloLimiter::new(MAX_HELLOS_PER_NODE_PER_SEC),
 136:             clock_skew_tolerance_secs,
 137:             max_sessions_per_node,
 138:         }
 139:     }
 140: 
 141:     /// Process a session establishment request.
 142:     ///
 143:     /// All security checks are performed in a deliberate order:
 144:     ///
 145:     /// 1. Hello rate limit (cheap, no crypto — shed load before signature work)
 146:     /// 2. Signature verification (ed25519, inherently constant-time)
 147:     /// 3. TTL bound check (max 120 s)
 148:     /// 4. Token freshness / expiry
 149:     /// 5. Replay nonce check
 150:     /// 6. `node_id` binding (ct_eq: hello.node_id == token.node_id)
 151:     /// 7. `peer_node_id` binding (ct_eq: hello.peer_node_id == token.peer_node_id)
 152:     /// 8. `relay_id` binding (ct_eq: token.relay_id == self.relay_id)
 153:     /// 9. Scope enforcement (token.scope == "forward_ciphertext_only")
 154:     /// 10. Per-node session capacity
 155:     pub fn handle_hello_from_tuple(
 156:         &mut self,
 157:         hello: RelayHello,
 158:         observed_addr: SocketAddr,
 159:     ) -> RelayHelloResponse {
 160:         // Check 1: Hello rate limit — shed before any crypto work
 161:         if !self.hello_limiter.check(&hello.node_id) {
 162:             return RelayHelloResponse::Rejected(RejectReason::RateLimitExceeded);
 163:         }
 164: 
 165:         // Check 2: Verify token signature (ed25519, constant-time internally)
 166:         if let Err(e) = hello
 167:             .session_token
 168:             .verify_signature(&self.control_verifier_key)
 169:         {
 170:             eprintln!("Relay hello rejected: invalid signature: {e}");
 171:             return RelayHelloResponse::Rejected(RejectReason::InvalidToken);
 172:         }
 173: 
 174:         // Check 3: Verify token TTL bound (max 120 s)
 175:         let ttl = hello.session_token.ttl_secs();
 176:         if ttl > MAX_RELAY_TTL_SECS {
 177:             eprintln!("Relay hello rejected: TTL exceeds max ({ttl} > {MAX_RELAY_TTL_SECS})");
 178:             return RelayHelloResponse::Rejected(RejectReason::InvalidToken);
 179:         }
 180: 
 181:         // Check 4: Token freshness / expiry
 182:         let now_unix = SystemTime::now()
 183:             .duration_since(UNIX_EPOCH)
 184:             .unwrap()
 185:             .as_secs();
 186: 
 187:         if hello
 188:             .session_token
 189:             .is_expired(now_unix, self.clock_skew_tolerance_secs)
 190:         {
 191:             eprintln!("Relay hello rejected: token expired");
 192:             return RelayHelloResponse::Rejected(RejectReason::ExpiredToken);
 193:         }
 194: 
 195:         // Check 5: Replay nonce
 196:         if self.nonce_store.contains(&hello.session_token.nonce) {
 197:             eprintln!("Relay hello rejected: nonce replay detected");
 198:             return RelayHelloResponse::Rejected(RejectReason::ReplayedNonce);
 199:         }
 200: 
 201:         // Check 6: node_id binding (constant-time)
 202:         let node_id_match: bool = hello
 203:             .node_id
 204:             .as_bytes()
 205:             .ct_eq(hello.session_token.node_id.as_bytes())
 206:             .into();
 207:         if !node_id_match {
 208:             eprintln!("Relay hello rejected: node_id mismatch between hello and token");
 209:             return RelayHelloResponse::Rejected(RejectReason::InvalidToken);
 210:         }
 211: 
 212:         // Check 7: peer_node_id binding (constant-time)
 213:         let peer_match: bool = hello
 214:             .peer_node_id
 215:             .as_bytes()
 216:             .ct_eq(hello.session_token.peer_node_id.as_bytes())
 217:             .into();
 218:         if !peer_match {
 219:             eprintln!("Relay hello rejected: peer_node_id mismatch");
 220:             return RelayHelloResponse::Rejected(RejectReason::PeerMismatch);
 221:         }
 222: 
 223:         // Check 8: relay_id binding (constant-time byte comparison)
 224:         let relay_match: bool = hello.session_token.relay_id.ct_eq(&self.relay_id).into();
 225:         if !relay_match {
 226:             eprintln!("Relay hello rejected: token relay_id does not match this relay");
 227:             return RelayHelloResponse::Rejected(RejectReason::InvalidToken);
 228:         }
 229: 
 230:         // Check 9: Scope enforcement
 231:         if hello.session_token.scope != RELAY_TOKEN_SCOPE {
 232:             eprintln!(
 233:                 "Relay hello rejected: unexpected scope '{}'",
 234:                 hello.session_token.scope
 235:             );
 236:             return RelayHelloResponse::Rejected(RejectReason::InvalidToken);
 237:         }
 238: 
 239:         // Check 10: Per-node session capacity
 240:         let node_session_count = self
 241:             .sessions
 242:             .values()
 243:             .filter(|s| s.node_id == hello.node_id)
 244:             .count();
 245: 
 246:         if node_session_count >= self.max_sessions_per_node {
 247:             eprintln!(
 248:                 "Relay hello rejected: capacity limit reached for node {}",
 249:                 hello.node_id
 250:             );
 251:             return RelayHelloResponse::Rejected(RejectReason::Capacity);
 252:         }
 253: 
 254:         // All checks passed — record nonce to prevent replay
 255:         self.nonce_store.insert(hello.session_token.nonce);
 256: 
 257:         // Allocate session
 258:         self.remove_session_for_pair(&hello.node_id, &hello.peer_node_id);
 259: 
 260:         let session_id = SessionId::generate();
 261:         let allocated_port = self.allocate_port();
 262: 
 263:         let session = RelaySession {
 264:             session_id,
 265:             node_id: hello.node_id.clone(),
 266:             peer_node_id: hello.peer_node_id.clone(),
 267:             allocated_port,
 268:             hello_source_addr: observed_addr,
 269:             bound_peer_addr: None,
 270:             expires_at_unix: hello.session_token.expires_at_unix,
 271:             established_at: Instant::now(),
 272:             last_packet_at: Instant::now(),
 273:         };
 274: 
 275:         self.sessions.insert(session_id, session);
 276:         self.node_pair_index.insert(
 277:             (hello.node_id.clone(), hello.peer_node_id.clone()),
 278:             session_id,
 279:         );
 280: 
 281:         RelayHelloResponse::Accepted(RelayHelloAck {
 282:             session_id,
 283:             allocated_port,
 284:         })
 285:     }

 293:     /// Refresh session activity timestamp without forwarding data.
 294:     ///
 295:     /// Used by keepalive packets to prevent session idle timeout.
 296:     pub fn touch_session_from_tuple(
 297:         &mut self,
 298:         session_id: SessionId,
 299:         from_addr: SocketAddr,
 300:     ) -> Result<bool, RelayForwardError> {
 301:         let now_unix = SystemTime::now()
 302:             .duration_since(UNIX_EPOCH)
 303:             .unwrap()
 304:             .as_secs();
 305: 
 306:         let Some(session) = self.sessions.get_mut(&session_id) else {
 307:             return Err(RelayForwardError::SessionNotFound);
 308:         };
 309:         if session.expires_at_unix <= now_unix {
 310:             self.remove_session(session_id);
 311:             return Err(RelayForwardError::SessionExpired);
 312:         }
 313:         match session.bound_peer_addr {
 314:             Some(bound_addr) if bound_addr == from_addr => {
 315:                 session.last_packet_at = Instant::now();
 316:                 Ok(true)
 317:             }
 318:             Some(_) => Err(RelayForwardError::UnauthorizedSourceTuple),
 319:             None => Ok(false),
 320:         }
 321:     }

 337:     /// Forward a ciphertext payload from one session to its paired session.
 338:     ///
 339:     /// Returns:
 340:     /// - `Ok(Some((peer_session_id, payload)))` — forwarded successfully
 341:     /// - `Ok(None)` — silently dropped (rate limit or no paired session)
 342:     /// - `Err(msg)` — session not found
 343:     ///
 344:     /// Payloads are forwarded byte-for-byte.  The relay never inspects content.
 345:     pub fn forward_packet(
 346:         &mut self,
 347:         session_id: SessionId,
 348:         payload: &[u8],
 349:         from_addr: SocketAddr,
 350:     ) -> Result<Option<RelayForwardTarget>, RelayForwardError> {
 351:         // Silently drop oversized payloads (do not error — avoids amplification)
 352:         if payload.len() > MAX_PACKET_SIZE_BYTES {
 353:             return Ok(None);
 354:         }
 355: 
 356:         let now_unix = SystemTime::now()
 357:             .duration_since(UNIX_EPOCH)
 358:             .unwrap()
 359:             .as_secs();
 360: 
 361:         let Some(session) = self.sessions.get(&session_id) else {
 362:             return Err(RelayForwardError::SessionNotFound);
 363:         };
 364:         if session.expires_at_unix <= now_unix {
 365:             self.remove_session(session_id);
 366:             return Err(RelayForwardError::SessionExpired);
 367:         }
 368: 
 369:         {
 370:             let session = self
 371:                 .sessions
 372:                 .get_mut(&session_id)
 373:                 .expect("session should remain available while forwarding");
 374:             match session.bound_peer_addr {
 375:                 Some(bound_addr) if bound_addr == from_addr => {}
 376:                 Some(_) => return Err(RelayForwardError::UnauthorizedSourceTuple),
 377:                 None => {
 378:                     if session.hello_source_addr.ip() != from_addr.ip() {
 379:                         return Err(RelayForwardError::UnauthorizedSourceTuple);
 380:                     }
 381:                     session.bound_peer_addr = Some(from_addr);
 382:                 }
 383:             }
 384: 
 385:             // Rate limit check — silent drop on excess
 386:             if !self
 387:                 .rate_limiter
 388:                 .check_packet(&session.node_id, payload.len())
 389:             {
 390:                 return Ok(None);
 391:             }
 392: 
 393:             // Update last_packet_at
 394:             session.last_packet_at = Instant::now();
 395:         }
 396: 
 397:         let (peer_session_id, current_node_id, current_peer_node_id) = {
 398:             let session = self
 399:                 .sessions
 400:                 .get(&session_id)
 401:                 .expect("session should remain available while forwarding");
 402:             (
 403:                 self.node_pair_index
 404:                     .get(&(session.peer_node_id.clone(), session.node_id.clone()))
 405:                     .copied(),
 406:                 session.node_id.clone(),
 407:                 session.peer_node_id.clone(),
 408:             )
 409:         };
 410: 
 411:         let Some(peer_sid) = peer_session_id else {
 412:             // Half-open: no paired session yet; silently drop
 413:             return Ok(None);
 414:         };
 415: 
 416:         let Some(peer_session) = self.sessions.get_mut(&peer_sid) else {
 417:             self.node_pair_index
 418:                 .remove(&(current_peer_node_id, current_node_id));
 419:             return Ok(None);
 420:         };
 421:         if peer_session.expires_at_unix <= now_unix {
 422:             self.remove_session(peer_sid);
 423:             return Ok(None);
 424:         }
 425:         let Some(peer_addr) = peer_session.bound_peer_addr else {
 426:             return Ok(None);
 427:         };
 428:         peer_session.last_packet_at = Instant::now();
 429:         // Forward payload as-is — never inspect content
 430:         Ok(Some(RelayForwardTarget {
 431:             peer_session_id: peer_sid,
 432:             peer_allocated_port: peer_session.allocated_port,
 433:             peer_addr,
 434:             payload: payload.to_vec(),
 435:         }))
 436:     }

 438:     /// Evict idle, half-open, and stale sessions, and prune the nonce store.
 439:     ///
 440:     /// Should be called periodically (e.g., every 10 seconds).
 441:     pub fn cleanup_idle_sessions(&mut self) -> Vec<RelaySession> {
 442:         let now = Instant::now();
 443:         let idle_threshold = Duration::from_secs(IDLE_SESSION_TIMEOUT_SECS);
 444:         let half_open_threshold = Duration::from_secs(HALF_OPEN_SESSION_TIMEOUT_SECS);
 445:         let now_unix = SystemTime::now()
 446:             .duration_since(UNIX_EPOCH)
 447:             .unwrap()
 448:             .as_secs();
 449: 
 450:         let mut to_remove = Vec::new();
 451: 
 452:         for (session_id, session) in &self.sessions {
 453:             let age = now.duration_since(session.established_at);
 454:             let idle = now.duration_since(session.last_packet_at);
 455: 
 456:             let has_pair = self
 457:                 .node_pair_index
 458:                 .contains_key(&(session.peer_node_id.clone(), session.node_id.clone()));
 459: 
 460:             // Remove session if:
 461:             // - it has expired according to the signed session token, OR
 462:             // - it is half-open (no paired session) and exceeds the half-open timeout, OR
 463:             // - it is idle (no recent packets) and exceeds the idle timeout
 464:             let is_expired = session.expires_at_unix <= now_unix;
 465:             let is_stale_half_open = !has_pair && age > half_open_threshold;
 466:             let is_idle = idle > idle_threshold;
 467:             if is_expired || is_stale_half_open || is_idle {
 468:                 to_remove.push(*session_id);
 469:             }
 470:         }
 471: 
 472:         let mut removed = Vec::new();
 473:         for sid in to_remove {
 474:             if let Some(session) = self.remove_session(sid) {
 475:                 removed.push(session);
 476:             }
 477:         }
 478: 
 479:         // Prune nonce store: remove entries older than the nonce retention window
 480:         // to prevent unbounded memory growth while keeping anti-replay guarantees.
 481:         self.nonce_store
 482:             .prune(Duration::from_secs(NONCE_RETENTION_SECS));
 483: 
 484:         removed
 485:     }

 487:     fn allocate_port(&self) -> u16 {
 488:         let base = 50_000u16;
 489:         let offset = (self.sessions.len() % 10_000) as u16;
 490:         base.wrapping_add(offset)
 491:     }

 523: }

 525: impl NonceStore {

 527:         self.nonces.contains_key(nonce)
 528:     }

 541: // ── Hello rate limiter ───
 542: 
 543: /// Limits session-establishment (`handle_hello`) attempts per node per second.
 544: ///
 545: /// This is a separate, much tighter limit than the packet-level `RateLimiter`
 546: /// because `handle_hello` performs ed25519 verification (CPU-intensive).
 547: /// Applying a cheap counter before verification protects against CPU-exhaustion
 548: /// attacks.
 549: struct HelloLimiter {
 550:     max_per_sec: u32,
 551:     counts: HashMap<String, (u32, Instant)>,
 552: }
 553: 
 554: impl HelloLimiter {
 555:     fn new(max_per_sec: u32) -> Self {
 556:         Self {
 557:             max_per_sec,
 558:             counts: HashMap::new(),
 559:         }
 560:     }
 561: 
 562:     /// Returns `true` if the hello should be allowed, `false` if rate-limited.
 563:     fn check(&mut self, node_id: &str) -> bool {
 564:         let now = Instant::now();
 565:         let entry = self.counts.entry(node_id.to_string()).or_insert((0, now));
 566: 
 567:         // Reset counter if the one-second window has elapsed
 568:         if now.duration_since(entry.1) >= Duration::from_secs(1) {
 569:             *entry = (0, now);
 570:         }
 571: 
 572:         if entry.0 >= self.max_per_sec {
 573:             return false;
 574:         }
 575:         entry.0 += 1;
 576:         true
 577:     }
 578: }
