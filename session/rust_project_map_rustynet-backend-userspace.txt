   1: //! Boringtun userspace WireGuard backend for Rustynet.
   2: //!
   3: //! Provides [`UserspaceBackend`], a second non-kernel TunnelBackend
   4: //! implementation that drives WireGuard entirely in userspace via the
   5: //! vendored boringtun noise engine. The kernel WireGuard module is not
   6: //! required; this backend can run on hosts where wg(8) is unavailable.
   7: //!
   8: //! On Linux this wraps [`LinuxUserspaceSharedBackend`] from
   9: //! `rustynet-backend-wireguard`. On other platforms the type is present but
  10: //! all operations fail with a clear `Internal` error so callers can detect
  11: //! platform capability at runtime rather than compile time.

  23: /// A boringtun-driven userspace WireGuard backend.
  24: ///
  25: /// On Linux this delegates all operations to

  27: /// call returns a `BackendError::internal` with a platform-unavailable

  32:     #[cfg(not(target_os = "linux"))]

  36: impl UserspaceBackend {
  37:     /// Construct a userspace backend for the given WireGuard interface.
  38:     ///
  39:     /// # Arguments
  40:     /// * `interface_name` – TUN interface name (e.g. `"rustynet0"`).
  41:     /// * `private_key_path` – Path to a base64-encoded WireGuard private key.
  42:     /// * `listen_port` – UDP port the userspace engine should bind.
  43:     ///
  44:     /// Returns an error if the platform does not support this backend or if
  45:     /// the constructor arguments are invalid.
  46:     pub fn new(
  47:         interface_name: impl Into<String>,
  48:         private_key_path: impl Into<String>,
  49:         listen_port: u16,
  50:     ) -> Result<Self, BackendError> {
  51:         #[cfg(target_os = "linux")]
  52:         {
  53:             let inner = LinuxUserspaceSharedBackend::new(
  54:                 interface_name,
  55:                 private_key_path,
  56:                 listen_port,
  57:             )?;
  58:             return Ok(Self { inner });
  59:         }
  60:         #[cfg(not(target_os = "linux"))]
  61:         {
  62:             let _ = (interface_name, private_key_path, listen_port);
  63:             Err(BackendError::internal(
  64:                 "UserspaceBackend is only available on Linux",
  65:             ))
  66:         }
  67:     }
  68: }
  69: 
  70: #[cfg(not(target_os = "linux"))]
  71: fn platform_unavailable() -> BackendError {

  19: #[test]
  20: fn userspace_backend_name_is_stable() {
  21: 
  22:     // On non-Linux the constructor returns an error; verify the name is
  23:     // accessible from a successfully-constructed instance only on Linux, and
  24:     // verify the error type on other platforms.
  25:     #[cfg(target_os = "linux")]
  26:     {
  27:         // Use a guaranteed-invalid key path so the constructor rejects it
  28:         // without touching the network — we still get a valid error type.
  29:         let result = UserspaceBackend::new("rustynet0", "/nonexistent/key.key", 51820);
  30:         match result {
  31:             Ok(mut b) => {
  32:                 assert_eq!(b.name(), "userspace-wireguard");
  33:                 let _ = b.shutdown();
  34:             }
  35:             Err(e) => {
  36:                 assert_eq!(
  37:                     e.kind,
  38:                     BackendErrorKind::InvalidInput,
  39:                     "invalid key path should return InvalidInput, got: {e}"
  40:                 );
  41:             }
  42:         }
  43:     }
  44: 
  45:     #[cfg(not(target_os = "linux"))]
  46:     {
  47:         let result = UserspaceBackend::new("rustynet0", "/any/key", 51820);
  48:         assert!(
  49:             result.is_err(),
  50:             "UserspaceBackend must fail on non-Linux platform"
  51:         );
  52:         let err = result.err().expect("already asserted is_err");
  53:         assert_eq!(err.kind, BackendErrorKind::Internal);
  54:     }
