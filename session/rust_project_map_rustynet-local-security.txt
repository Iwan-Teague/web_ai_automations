 146: fn validate_root_managed_shared_runtime_socket_facts(
 147:     path: &Path,
 148:     label: &str,
 149:     facts: SocketSecurityFacts,
 150:     allowed_socket_owner_uids: &[u32],
 151:     allowed_parent_owner_uids: &[u32],
 152:     expected_gid: u32,
 153: ) -> Result<(), String> {
 154:     if facts.socket_mode & 0o007 != 0 {
 155:         return Err(format!(
 156:             "{label} permissions too broad ({:03o}); world access is forbidden: {}",
 157:             facts.socket_mode,
 158:             path.display()
 159:         ));
 160:     }
 161:     let root_managed_group_socket =
 162:         facts.socket_uid == 0 && facts.socket_gid == expected_gid && facts.socket_mode == 0o660;
 163:     if !owner_allowed(facts.socket_uid, allowed_socket_owner_uids) && !root_managed_group_socket {
 164:         return Err(format!(
 165:             "{label} owner uid mismatch: allowed {:?}, found {} ({})",
 166:             allowed_socket_owner_uids,
 167:             facts.socket_uid,
 168:             path.display()
 169:         ));
 170:     }
 171:     if facts.socket_mode & 0o070 != 0
 172:         && !root_managed_group_socket
 173:         && facts.socket_gid != expected_gid
 174:     {
 175:         return Err(format!(
 176:             "{label} group mismatch: expected gid {expected_gid}, found {} ({})",
 177:             facts.socket_gid,
 178:             path.display()
 179:         ));
 180:     }
 181: 
 182:     if facts.parent_mode & 0o002 != 0 {
 183:         return Err(format!(
 184:             "{label} parent directory has insecure permissions: mode {:o}",
 185:             facts.parent_mode
 186:         ));
 187:     }
 188:     let root_managed_shared_runtime =
 189:         facts.parent_uid == 0 && facts.parent_gid == expected_gid && facts.parent_mode == 0o770;
 190:     if !owner_allowed(facts.parent_uid, allowed_parent_owner_uids) && !root_managed_shared_runtime {
 191:         return Err(format!(
 192:             "{label} parent directory owner uid mismatch: allowed {allowed_parent_owner_uids:?}, found {}",
 193:             facts.parent_uid,
 194:         ));
 195:     }
 196:     if facts.parent_mode & 0o020 != 0
 197:         && !root_managed_shared_runtime
 198:         && facts.parent_gid != expected_gid
 199:     {
 200:         return Err(format!(
 201:             "{label} parent directory group mismatch: expected gid {expected_gid}, found {}",
 202:             facts.parent_gid
