 544:         let fqdn = format!("{label}.{zone_name}");
 545:         if !seen_names.insert(fqdn.clone()) {
 546:             return Err(DnsZoneError::InvalidFormat(
 547:                 "duplicate dns record name".to_string(),
 548:             ));
 549:         }
 550:         let target_node_id = record.target_node_id.trim().to_string();
 551:         if target_node_id.is_empty() {
 552:             return Err(DnsZoneError::InvalidFormat(
 553:                 "target_node_id must not be empty".to_string(),
 554:             ));
 555:         }
 556:         let expected_ip = parse_expected_ip(
 557:             record.expected_ip.as_str(),
 558:             record.target_addr_kind,
 559:             canonical_records.len(),
 560:         )?;
 561:         let mut aliases = record
 562:             .aliases
 563:             .iter()
 564:             .map(|alias| canonicalize_dns_relative_name(alias))
 565:             .collect::<Result<Vec<_>, _>>()?;
 566:         aliases.sort();
 567:         aliases.dedup();
 568:         if aliases.len() > MAX_ALIAS_COUNT {
 569:             return Err(DnsZoneError::InvalidFormat(format!(
 570:                 "dns record exceeds maximum alias count ({MAX_ALIAS_COUNT})"
 571:             )));
 572:         }
 573:         for alias in &aliases {
 574:             let alias_fqdn = format!("{alias}.{zone_name}");
 575:             if !seen_names.insert(alias_fqdn) {
 576:                 return Err(DnsZoneError::InvalidFormat(
 577:                     "dns alias collides with another record".to_string(),
 578:                 ));
 579:             }
 580:         }
 581:         canonical_records.push(DnsZoneRecord {
 582:             label,
 583:             fqdn,
 584:             target_node_id,
 585:             rr_type: record.rr_type,
 586:             target_addr_kind: record.target_addr_kind,
 587:             expected_ip,
 588:             ttl_secs: record.ttl_secs,
 589:             aliases,
 590:         });
 591:     }
 592:     canonical_records.sort_by(|left, right| left.fqdn.cmp(&right.fqdn));
 593:     Ok(canonical_records)
 594: }

 596: fn serialize_dns_zone_payload(
 597:     zone_name: &str,
 598:     subject_node_id: &str,
 599:     generated_at_unix: u64,
 600:     expires_at_unix: u64,
 601:     nonce: u64,
 602:     records: &[DnsZoneRecord],
 603: ) -> String {
 604:     let mut payload = String::new();
 605:     payload.push_str("version=1\n");
 606:     payload.push_str(&format!("zone_name={zone_name}\n"));
 607:     payload.push_str(&format!("subject_node_id={subject_node_id}\n"));
 608:     payload.push_str(&format!("generated_at_unix={generated_at_unix}\n"));
 609:     payload.push_str(&format!("expires_at_unix={expires_at_unix}\n"));
 610:     payload.push_str(&format!("nonce={nonce}\n"));
 611:     payload.push_str(&format!("record_count={}\n", records.len()));
 612:     for (index, record) in records.iter().enumerate() {
 613:         payload.push_str(&format!("record.{index}.label={}\n", record.label));
 614:         payload.push_str(&format!("record.{index}.fqdn={}\n", record.fqdn));
 615:         payload.push_str(&format!(
 616:             "record.{index}.target_node_id={}\n",
 617:             record.target_node_id
 618:         ));
 619:         payload.push_str(&format!(
 620:             "record.{index}.rr_type={}\n",
 621:             record.rr_type.as_str()
 622:         ));
 623:         payload.push_str(&format!(
 624:             "record.{index}.target_addr_kind={}\n",
 625:             record.target_addr_kind.as_str()
 626:         ));
 627:         payload.push_str(&format!(
 628:             "record.{index}.expected_ip={}\n",
 629:             record.expected_ip
 630:         ));
 631:         payload.push_str(&format!("record.{index}.ttl_secs={}\n", record.ttl_secs));
 632:         payload.push_str(&format!(
 633:             "record.{index}.aliases={}\n",
 634:             record.aliases.join(",")
 635:         ));
 636:     }
 637:     payload
 638: }

 640: fn parse_expected_ip(
 641:     value: &str,
 642:     target_addr_kind: DnsTargetAddrKind,
 643:     index: usize,
 644: ) -> Result<String, DnsZoneError> {
 645:     match target_addr_kind {
 646:         DnsTargetAddrKind::MeshIpv4 => {
 647:             let ip = value.parse::<std::net::Ipv4Addr>().map_err(|_| {
 648:                 DnsZoneError::InvalidFormat(format!("record {index} expected_ip must be ipv4"))
 649:             })?;
 650:             if ip.is_unspecified() || ip.is_multicast() || ip.is_broadcast() {
 651:                 return Err(DnsZoneError::InvalidFormat(format!(
 652:                     "record {index} expected_ip must be a unicast ipv4 address"
 653:                 )));
 654:             }
 655:             // Defense-in-depth (related to SecurityHardeningAudit B.4.1):
 656:             // a malicious zone-publisher must not be able to inject an
 657:             // expected_ip in a range that is universally-inappropriate
 658:             // for a mesh peer. The daemon's loopback resolver answer
 659:             // filter (B.4.1 proper) is still pending — once the
 660:             // protocol-level DNS responder lands the same posture
 661:             // applies at the resolver-output layer too. Until then,
 662:             // catching these at zone-bundle parse time is the strictest
 663:             // defense the signed-state contract can publish.
 664:             if is_universally_inappropriate_mesh_ipv4(&ip) {
 665:                 return Err(DnsZoneError::InvalidFormat(format!(
 666:                     "record {index} expected_ip {} is in a range that cannot host a mesh peer (loopback, link-local, or RFC 5737 documentation)",
 667:                     ip
 668:                 )));
 669:             }
 670:             Ok(ip.to_string())
 671:         }
 672:     }
 673: }
 674: 
 675: /// Returns true for IPv4 ranges that are universally inappropriate as
 676: /// a mesh-peer expected_ip — irrespective of operator network choices.
 677: /// The list is intentionally narrow: loopback, link-local APIPA, and
 678: /// the three RFC 5737 documentation ranges. RFC1918 ranges (10/8,
 679: /// 172.16-31/16, 192.168/16) are NOT rejected here because some
 680: /// operators legitimately deploy meshes inside their corporate RFC1918
 681: /// space; rejecting RFC1918 globally would break those deployments.
 682: /// An operator who wants to additionally constrain the mesh IP range
 683: /// to a specific allowlist publishes that constraint via a separate
 684: /// per-deployment policy gate (out of scope for the zone-bundle
 685: /// validator, which intentionally publishes only universally-true
 686: /// invariants).
 687: fn is_universally_inappropriate_mesh_ipv4(ip: &std::net::Ipv4Addr) -> bool {
 688:     let octets = ip.octets();
 689:     // Loopback: 127.0.0.0/8 — RFC 6890. Never a mesh peer.
 690:     if octets[0] == 127 {
 691:         return true;
 692:     }
 693:     // Link-local APIPA: 169.254.0.0/16 — RFC 3927.
 694:     if octets[0] == 169 && octets[1] == 254 {
 695:         return true;
 696:     }
 697:     // RFC 5737 documentation / TEST-NET-{1,2,3}.
 698:     // 192.0.2.0/24
 699:     if octets[0] == 192 && octets[1] == 0 && octets[2] == 2 {
 700:         return true;
 701:     }
 702:     // 198.51.100.0/24
 703:     if octets[0] == 198 && octets[1] == 51 && octets[2] == 100 {
 704:         return true;
 705:     }
 706:     // 203.0.113.0/24
 707:     if octets[0] == 203 && octets[1] == 0 && octets[2] == 113 {
 708:         return true;
 709:     }
 710:     false
 711: }

 713: fn parse_aliases(value: &str, zone_name: &str, index: usize) -> Result<Vec<String>, DnsZoneError> {
 714:     if value.trim().is_empty() {
 715:         return Ok(Vec::new());
 716:     }
 717:     let mut aliases = value
 718:         .split(',')
 719:         .map(str::trim)
 720:         .filter(|alias| !alias.is_empty())
 721:         .map(canonicalize_dns_relative_name)
 722:         .collect::<Result<Vec<_>, _>>()?;
 723:     aliases.sort();
 724:     aliases.dedup();
 725:     for alias in &aliases {
 726:         let alias_fqdn = format!("{alias}.{zone_name}");
 727:         if alias_fqdn.len() > 253 {
 728:             return Err(DnsZoneError::InvalidFormat(format!(
 729:                 "record {index} alias exceeds maximum fqdn length"
 730:             )));
 731:         }
 732:     }
 733:     Ok(aliases)
 734: }

 784: fn record_indexed_key(key: &str) -> Option<(&str, usize)> {
 785:     let mut parts = key.split('.');
 786:     let root = parts.next()?;
 787:     if root != "record" {
 788:         return None;
 789:     }
 790:     let index = parts.next()?.parse::<usize>().ok()?;
 791:     let field = parts.next()?;
 792:     if parts.next().is_some() {
 793:         return None;
 794:     }
 795:     match field {
 796:         "label" | "fqdn" | "target_node_id" | "rr_type" | "target_addr_kind" | "expected_ip"
 797:         | "ttl_secs" | "aliases" => Some((field, index)),
 798:         _ => None,
 799:     }
 800: }

 802: fn sha256_digest(payload: &[u8]) -> [u8; 32] {
 803:     let mut hasher = Sha256::new();
 804:     hasher.update(payload);
 805:     let digest = hasher.finalize();
 806:     let mut out = [0u8; 32];
 807:     out.copy_from_slice(&digest[..32]);
 808:     out
 809: }

 812:     let mut encoded = String::with_capacity(bytes.len() * 2);
 813:     for byte in bytes {

 815:     }

 819: fn decode_hex_to_fixed<const N: usize>(encoded: &str) -> Result<[u8; N], DnsZoneError> {
 820:     let trimmed = encoded.trim();
 821:     if trimmed.len() != N * 2 {
 822:         return Err(DnsZoneError::InvalidFormat(
 823:             "hex value has invalid length".to_string(),
 824:         ));
 825:     }
 826:     let raw = trimmed.as_bytes();
 827:     let mut bytes = [0u8; N];
 828:     for index in 0..N {
 829:         let hi = decode_hex_nibble(raw[index * 2])?;
 830:         let lo = decode_hex_nibble(raw[index * 2 + 1])?;
 831:         bytes[index] = (hi << 4) | lo;
 832:     }
 833:     Ok(bytes)
 834: }
 835: 
 836: fn decode_hex_nibble(value: u8) -> Result<u8, DnsZoneError> {
 837:     match value {
 838:         b'0'..=b'9' => Ok(value - b'0'),
 839:         b'a'..=b'f' => Ok(value - b'a' + 10),
 840:         b'A'..=b'F' => Ok(value - b'A' + 10),
 841:         _ => Err(DnsZoneError::InvalidFormat(
 842:             "hex value contains invalid character".to_string(),
 843:         )),
 844:     }
 845: }
