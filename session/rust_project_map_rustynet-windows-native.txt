 295:             CreateNamedPipeW(
 296:                 wide_path.as_ptr(),

 301:                 message_bytes,

 305:         };
 306:         let handle = OwnedHandle::new(handle).ok_or_else(|| {
 307:             format!("CreateNamedPipeW failed with Windows error {}", unsafe {

 311: 
 312:         let connected = unsafe { ConnectNamedPipe(handle.raw(), null_mut()) };
 313:         if connected == 0 {

 595:     fn inspect_handle_sddl(handle: HANDLE) -> Result<String, String> {
 596:         let requested = OWNER_SECURITY_INFORMATION | DACL_SECURITY_INFORMATION;
 597:         let mut needed = 0u32;
 598:         let first =
 599:             unsafe { GetKernelObjectSecurity(handle, requested, null_mut(), 0, &mut needed) };
 600:         if first != 0 {
 601:             return Err(
 602:                 "GetKernelObjectSecurity unexpectedly succeeded with a zero-length buffer"
 603:                     .to_string(),
 604:             );
 605:         }
 606:         let err = unsafe { GetLastError() };
 607:         if err != ERROR_INSUFFICIENT_BUFFER || needed == 0 {
 608:             return Err(format!(
 609:                 "GetKernelObjectSecurity sizing failed with Windows error {err}"
 610:             ));
 611:         }
 612:         let mut buffer = vec![0u8; needed as usize];
 613:         let ok = unsafe {
 614:             GetKernelObjectSecurity(
 615:                 handle,
 616:                 requested,
 617:                 buffer.as_mut_ptr().cast::<c_void>(),
 618:                 needed,
 619:                 &mut needed,
 620:             )
 621:         };
 622:         if ok == 0 {
 623:             return Err(format!(
 624:                 "GetKernelObjectSecurity failed with Windows error {}",
 625:                 unsafe { GetLastError() }
 626:             ));
 627:         }
 628:         security_descriptor_to_sddl(buffer.as_mut_ptr().cast::<c_void>())
 629:     }

 631:     /// Authenticode chain validation via `WinVerifyTrust`. Wraps the
 632:     /// canonical `WINTRUST_ACTION_GENERIC_VERIFY_V2` verb with
 633:     /// chain-revocation checking enabled (`WTD_REVOCATION_CHECK_CHAIN_EXCLUDE_ROOT`)
 634:     /// and the UI suppressed (`WTD_UI_NONE`). Verification succeeds
 635:     /// only when the full PKCS#7 chain is trusted, the file digest
 636:     /// matches the SpcIndirectData, and any counter-signature
 637:     /// timestamps are valid; any of those failing maps to a typed
 638:     /// `Untrusted` outcome with the canonical HRESULT label.
 639:     ///
 640:     /// State cleanup: the Win32 contract requires a follow-up call
 641:     /// with `dwStateAction = WTD_STATEACTION_CLOSE` after every
 642:     /// `WTD_STATEACTION_VERIFY`. The wrapper performs the cleanup
 643:     /// even when the verify call returns an error so we do not leak
 644:     /// the verifier state into subsequent calls.
 645:     pub fn verify_authenticode_chain(
 646:         path: &Path,
 647:     ) -> Result<super::AuthenticodeChainOutcome, String> {
 648:         use std::os::windows::ffi::OsStrExt;
 649:         use windows_sys::Win32::Security::WinTrust::{
 650:             WINTRUST_ACTION_GENERIC_VERIFY_V2, WINTRUST_DATA, WINTRUST_DATA_0, WINTRUST_FILE_INFO,
 651:             WTD_CHOICE_FILE, WTD_REVOCATION_CHECK_CHAIN_EXCLUDE_ROOT, WTD_STATEACTION_CLOSE,
 652:             WTD_STATEACTION_VERIFY, WTD_UI_NONE, WinVerifyTrust,
 653:         };
 654: 
 655:         // Reject empty paths up-front — passing an empty string to
 656:         // WinVerifyTrust would surface as a confusing internal error
 657:         // rather than a clear "bad input" rejection.
 658:         if path.as_os_str().is_empty() {
 659:             return Err("verify_authenticode_chain: path must not be empty".to_string());
 660:         }
 661: 
 662:         // Encode path as UTF-16 with explicit NUL terminator. The
 663:         // Win32 wide-string contract requires the caller to provide
 664:         // the NUL.
 665:         let mut wide: Vec<u16> = path.as_os_str().encode_wide().collect();
 666:         wide.push(0);
 667: 
 668:         let mut file_info = WINTRUST_FILE_INFO {
 669:             cbStruct: size_of::<WINTRUST_FILE_INFO>() as u32,
 670:             pcwszFilePath: wide.as_ptr(),
 671:             hFile: null_mut(),
 672:             pgKnownSubject: null_mut(),
 673:         };
 674: 
 675:         let mut wvt_data = WINTRUST_DATA {
 676:             cbStruct: size_of::<WINTRUST_DATA>() as u32,
 677:             pPolicyCallbackData: null_mut(),
 678:             pSIPClientData: null_mut(),
 679:             dwUIChoice: WTD_UI_NONE,
 680:             fdwRevocationChecks: WTD_REVOCATION_CHECK_CHAIN_EXCLUDE_ROOT,
 681:             dwUnionChoice: WTD_CHOICE_FILE,
 682:             Anonymous: WINTRUST_DATA_0 {
 683:                 pFile: &mut file_info,
 684:             },
 685:             dwStateAction: WTD_STATEACTION_VERIFY,
 686:             hWVTStateData: null_mut(),
 687:             pwszURLReference: null_mut(),
 688:             dwProvFlags: 0,
 689:             dwUIContext: 0,
 690:             pSignatureSettings: null_mut(),
 691:         };
 692: 
 693:         let mut action_guid = WINTRUST_ACTION_GENERIC_VERIFY_V2;
 694:         let verify_status = unsafe {
 695:             WinVerifyTrust(
 696:                 null_mut(),
 697:                 &mut action_guid,
 698:                 &mut wvt_data as *mut WINTRUST_DATA as *mut c_void,
 699:             )
 700:         };
 701: 
 702:         // Always cleanup state, even on verify failure, to avoid
 703:         // leaking handles/state into subsequent calls.
 704:         wvt_data.dwStateAction = WTD_STATEACTION_CLOSE;
 705:         let _ = unsafe {
 706:             WinVerifyTrust(
 707:                 null_mut(),
 708:                 &mut action_guid,
 709:                 &mut wvt_data as *mut WINTRUST_DATA as *mut c_void,
 710:             )
 711:         };
 712: 
 713:         if verify_status == 0 {
 714:             Ok(super::AuthenticodeChainOutcome::Verified)
 715:         } else {
 716:             let raw_u32 = verify_status as u32;
 717:             let label = match raw_u32 {
 718:                 0x800B0001 => "TRUST_E_PROVIDER_UNKNOWN",
 719:                 0x800B0100 => "TRUST_E_NOSIGNATURE",
 720:                 0x800B0101 => "CERT_E_EXPIRED",
 721:                 0x800B0102 => "CERT_E_VALIDITYPERIODNESTING",
 722:                 0x800B0103 => "CERT_E_ROLE",
 723:                 0x800B0104 => "CERT_E_PATHLENCONST",
 724:                 0x800B0105 => "CERT_E_CRITICAL",
 725:                 0x800B0106 => "CERT_E_PURPOSE",
 726:                 0x800B0107 => "CERT_E_ISSUERCHAINING",
 727:                 0x800B0108 => "CERT_E_MALFORMED",
 728:                 0x800B0109 => "CERT_E_UNTRUSTEDROOT",
 729:                 0x800B010A => "CERT_E_CHAINING",
 730:                 0x800B010B => "TRUST_E_FAIL",
 731:                 0x800B010C => "CERT_E_REVOKED",
 732:                 0x800B010D => "CERT_E_UNTRUSTEDTESTROOT",
 733:                 0x800B010E => "CERT_E_REVOCATION_FAILURE",
 734:                 0x80092010 => "CRYPT_E_REVOKED",
 735:                 0x80092012 => "CRYPT_E_NO_REVOCATION_CHECK",
 736:                 0x80092013 => "CRYPT_E_REVOCATION_OFFLINE",
 737:                 0x80096010 => "TRUST_E_BAD_DIGEST",
 738:                 0x800B0110 => "TRUST_E_CERT_SIGNATURE",
 739:                 0x800B0111 => "TRUST_E_TIME_STAMP",
 740:                 _ => "UNKNOWN",
 741:             };
 742:             let reason = format!(
 743:                 "WinVerifyTrust returned 0x{raw_u32:08x} ({label}) for {}",
 744:                 path.display()
 745:             );
 746:             Ok(super::AuthenticodeChainOutcome::Untrusted {
 747:                 reason,
 748:                 hresult: verify_status as i64,
 749:             })
 750:         }
 751:     }
 752: }
